/*
 * Finished tasks' folders, removed to free the disk (ADR-056). A task is
 * finished once it is in Settled: done or abandoned, and not a merged one
 * with commits after its pull request. Its folders go after a wait that
 * suits how it ended, as the person chose: Smart to start, a review's at
 * once, a merged change's after 3 days, a done task's after a week, an
 * abandoned one's after 2 weeks, and any of them sooner when the disk runs
 * low; or one wait for all; or only when the person cleans up. The wait
 * runs from the latest of when it settled, when its screen was opened and
 * when its folder was opened from Althar.
 *
 * Nothing that exists only in a folder goes by itself: a folder with
 * changes not committed, an abandoned change's commits pushed nowhere,
 * something running from it, or opened in an editor within the day, or one
 * git can't read, is held back, and the person can still remove it by hand,
 * but while something runs from it. What holds a folder is asked again just
 * before it goes, while nothing can start on its thread, bring it back, open
 * it or start running from it (`withFolder`); by hand, it stays where more
 * would be lost with it than the person was told. A folder goes only once
 * its last commit is kept under `refs/althar/kept/<workspace>`, so its
 * changes still read and it comes back from there (`bringBack.ts`), and
 * counts as gone only once nothing of it is left; its branch stays. An idle
 * lead still held on it stops first. Its thread says it went, as a line.
 *
 * What each folder takes on the disk, and what holds it, is measured in
 * the background as Settings asks, and kept a while. A sweep runs a while
 * after Althar starts, every hour after, and at once when the person
 * chooses another wait.
 */

/**
 * On APFS, what a task's folders hold that nothing outside them shares, in
 * bytes, by getattrlist through the perl every Mac has. bun installs
 * `node_modules` as clones of its cache: `du` and Finder count them in full,
 * but removing them frees next to nothing. A file counts its private size
 * (ATTR_CMNEXT_PRIVATESIZE), what a package built or changed after it was
 * installed. Blocks a clone shares count nothing, even where every clone of
 * them is among the folders: APFS can't say so, as editing a clone gives it
 * a clone ID of its own and drops it from its source's count while it still
 * shares their blocks. So clones made of each other inside the folders count
 * short, never what removing them doesn't free. A file hardlinked from
 * outside them counts nothing; one whose links are all inside, once. What it
 * can't read fails it: a part is not the whole.
 */
const PRIVATE = String.raw`
use strict; use warnings; use File::Find;
my $list = pack('SSLLLLL', 5, 0, 0x80000000, 0, 0, 0, 0x8);
my ($total, $failed, %seen) = (0, 0);
local $SIG{__WARN__} = sub { $failed = 1 };
find({ no_chdir => 1, wanted => sub {
  my @st = lstat $_;
  unless (@st) { $failed = 1 unless $!{ENOENT}; return }
  my $own = $st[12] * 512;
  if (-f _) {
    return if $st[3] > 1 && ++$seen{"$st[0]:$st[1]"} != $st[3];
    my $buf = "\0" x 64;
    if (syscall(220, $_, $list, $buf, 64, 0x21) == 0 && unpack('L', $buf) >= 32) {
      $own = unpack('q', substr($buf, 24, 8));
    }
  }
  $total += $own;
}}, @ARGV);
exit 1 if $failed;
print "$total\n";
`
 
/**
 * What folders' files hold that nothing outside them shares, in bytes: a file
 * hardlinked from outside them, as bun and pnpm install `node_modules` on
 * Linux and Windows, counts nothing; one whose links are all inside, once.
 * Where the disk says no blocks, a file's length. What it can't read throws:
 * a part is not the whole.
 */
export const ownedBytes = async (roots: ReadonlyArray<string>): Promise<number> => {
  let total = 0
  const seen = new Map<string, number>()
  const gone = (error: unknown) => {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error
    return null
  }
  const walk = async (path: string): Promise<void> => {
    const stats = await lstat(path, { bigint: true }).catch(gone)
    if (stats === null) return
    if (stats.isDirectory()) {
      for (const name of (await readdir(path).catch(gone)) ?? []) await walk(join(path, name))
    }
    if (stats.nlink > 1n && stats.isFile()) {
      const key = `${stats.dev}:${stats.ino}`
      const count = (seen.get(key) ?? 0) + 1
      seen.set(key, count)
      if (BigInt(count) !== stats.nlink) return
    }
    total += Number(stats.blocks > 0n ? stats.blocks * 512n : stats.isFile() ? stats.size : 0n)
  }
  for (const root of roots) await walk(root)
  return total
}

/** A size for the thread: 2.3 GB, 14 GB, 640 MB, under 1 MB. */
export const sizeWords = (bytes: number) => {
  const gb = bytes / 1024 ** 3

/** Fails unless nothing of a folder is left. */
const gone = (path: string) => Effect.suspend(() => (existsSync(path) ? Effect.fail(new FolderStayed({ path })) : Effect.void))
 
/**
 * Removes a folder that isn't a worktree, or one git won't remove: whatever is in it. Fails unless nothing of it is left.

 * scanner reading what was just written, still holds it a moment.

const erase = (path: string) =>
  removeFully(path).pipe(
    Effect.mapError(() => new FolderStayed({ path })),
    Effect.andThen(gone(path)),

const eraseEmpty = (path: string) =>
